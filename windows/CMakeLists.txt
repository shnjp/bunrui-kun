cmake_minimum_required(VERSION 3.20)
project(bunrui_kun LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Build the Windows implementation on Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(router src/router.cpp)
target_include_directories(router PUBLIC src)
target_include_directories(router SYSTEM PRIVATE vendor)
target_compile_definitions(router PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
  target_compile_options(router PRIVATE /W4 /utf-8)
endif()
add_executable(bunrui-kun WIN32 src/main.cpp)
target_link_libraries(bunrui-kun PRIVATE router shell32 ole32 advapi32 user32 winhttp uuid)
if(MINGW)
  target_link_options(bunrui-kun PRIVATE -municode -static)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(router-tests tests/router_tests.cpp)
  target_link_libraries(router-tests PRIVATE router shell32)
  if(MINGW)
    target_link_options(router-tests PRIVATE -static)
  endif()
  add_test(NAME router-tests COMMAND router-tests)
  add_executable(capture-args WIN32 tests/capture_args.cpp)
  target_link_libraries(capture-args PRIVATE router shell32)
  if(MINGW)
    target_link_options(capture-args PRIVATE -municode -static)
  endif()
  add_test(NAME integration COMMAND powershell -NoProfile -ExecutionPolicy Bypass
    -File ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.ps1
    -Exe $<TARGET_FILE:bunrui-kun> -CaptureExe $<TARGET_FILE:capture-args>
    -Sample ${CMAKE_CURRENT_SOURCE_DIR}/config.example.toml)
endif()
# Do not install an active portable config: the user config must remain discoverable.
install(TARGETS bunrui-kun RUNTIME DESTINATION .)
install(FILES config.example.toml README.txt vendor/tomlplusplus-LICENSE DESTINATION .)
if(MINGW)
  install(FILES vendor/llvm-LICENSE.txt DESTINATION .)
endif()
